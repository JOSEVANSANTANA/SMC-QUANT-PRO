# No macOS a TIGER FALA pelo comando `say` do próprio sistema (nativo, não

# no código e porque no Windows é ele quem fala.
pyttsx3>=2.90

# Comando por voz ("Olá Tiger"). Opcional: sem eles o programa funciona
# inteiro e o botão do microfone explica como habilitar.
SpeechRecognition>=3.10.0

# O sounddevice IMPORTA numpy assim que é carregado. Sem esta linha o Mac
# respondia "No module named 'numpy'" e o microfone e a fala da TIGER ficavam
# mortos, sem que nada na tela explicasse por quê — o pip instalava o
# sounddevice, dava tudo certo, e o import falhava na hora de usar.
numpy>=1.24.0

# ------------------------------------------------------------------
# JANELAS DO macOS  (ESTE É O ITEM QUE NÃO PODE FALTAR)

# `screencapture`, que já vem no macOS.
pyobjc-framework-Quartz>=10.0
# O AppKit (Cocoa) diz QUAIS processos sao aplicativos de verdade — os que tem
# icone no Dock. E o que separa a janela do Chrome de um servico interno do
# macOS: sem ele, o seletor de janelas mostrava coisas como
# "Accessibility Services", que nunca vao ser o grafico de ninguem.
pyobjc-framework-Cocoa>=10.0
